#### Create coeffs.py and integrate.py. Flat numpy only — no classes, no config objects, minimal functions.  
####   
#### coeffs.py: one function transfer_matrix(alpha_I, alpha_Q, alpha_U, alpha_V, rho_Q, rho_U, rho_V) returning the 4x4 numpy matrix K for the polarized radiative transfer equation dS/ds = j - K @ S, where S = [I, Q, U, V]. Use this exact sign convention:  
#### row I: [alpha_I, alpha_Q, alpha_U, alpha_V]  
#### row Q: [alpha_Q, alpha_I, rho_V, -rho_U]  
#### row U: [alpha_U, -rho_V, alpha_I, rho_Q]  
#### row V: [alpha_V, rho_U, -rho_Q, alpha_I]  
####   
#### integrate.py: one function integrate_ray(s_grid, j_of_s, K_of_s, S0) doing fixed-step RK4 for dS/ds = j(s) - K(s) @ S along the given grid. j_of_s returns a length-4 array, K_of_s returns 4x4. Return the full solution array, shape (len(s_grid), 4). No adaptive stepping, no scipy.  

coeffs.py
```
import numpy as np

def transfer_matrix(alpha_I, alpha_Q, alpha_U, alpha_V, rho_Q, rho_U, rho_V):
    K = np.array([
        [alpha_I, alpha_Q, alpha_U, alpha_V],
        [alpha_Q, alpha_I, rho_V, -rho_U],
        [alpha_U, -rho_V, alpha_I, rho_Q],
        [alpha_V, rho_U, -rho_Q, alpha_I]
    ])
    return K
```

integrate.py
```
import numpy as np

def integrate_ray(s_grid, j_of_s, K_of_s, S0):
    def rk4_step(S, s, h, j, K):
        k1 = h * (j(s) - K(s) @ S)
        k2 = h * (j(s + 0.5*h) - K(s + 0.5*h) @ (S + 0.5*k1))
        k3 = h * (j(s + 0.5*h) - K(s + 0.5*h) @ (S + 0.5*k2))
        k4 = h * (j(s + h) - K(s + h) @ (S + k3))
        return S + (k1 + 2*k2 + 2*k3 + k4) / 6

    S = np.zeros((len(s_grid), 4))
    S[0] = S0
    for i in range(1, len(s_grid)):
        h = s_grid[i] - s_grid[i-1]
        S[i] = rk4_step(S[i-1], s_grid[i-1], h, j_of_s, K_of_s)
    
    return S
```

#### Setup: s from 0 to 1, 2000 steps. Region s < 0.3: emission only, j = [1, 0.7, 0, 0] (polarized at angle 0). Region s >= 0.3: pure Faraday rotation, rho_V = 2 * RM_TOTAL/0.7 * lam2 (so the column integrates to a total rotation of RM_TOTAL * lam2), with RM_TOTAL = 10.0. All absorption zero.  
####   
#### Sweep lam2 over [0.0, 0.02, 0.05, 0.1, 0.15]. For each, integrate, compute chi = 0.5*arctan2(U, Q) at s=1, print lam2, chi, and chi/lam2.

#### PASS criteria to print at the end:  
#### 1. chi/lam2 constant across the sweep and equal to 10.0 within 0.1%.

```
import numpy as np
from coeffs import transfer_matrix
from integrate import integrate_ray

# Constants
RM_TOTAL = 10.0
S0 = [1, 0, 0, 0]
s_grid_2000 = np.linspace(0, 1, 2000)
s_grid_1000 = np.linspace(0, 1, 1000)

# Emission region j(s) for s < 0.3
def j_of_s_emission(s):
    if s < 0.3:
        return np.array([1, 0.7, 0, 0])
    else:
        return np.zeros(4)

# Faraday rotation region K(s) for s >= 0.3
def K_of_s_faraday(s, lam2):
    rho_V = 2 * RM_TOTAL / 0.7 * lam2
    if s < 0.3:
        return transfer_matrix(0, 0, 0, 0, 0, 0, 0)
    else:
        return transfer_matrix(0, 0, 0, 0, 0, 0, rho_V)

# Sweep over lam2 values
lam2_values = [0.0, 0.02, 0.05, 0.1, 0.15]
results = []

for lam2 in lam2_values:
    K_of_s = lambda s: K_of_s_faraday(s, lam2)
    S_2000 = integrate_ray(s_grid_2000, j_of_s_emission, K_of_s, S0)
    chi = 0.5 * np.arctan2(S_2000[-1, 2], S_2000[-1, 1])
    results.append((lam2, chi, chi / lam2))
    print(f"lam2: {lam2}, chi: {chi}, chi/lam2: {chi / lam2}")

# Check PASS criteria
chi_over_lam2_values = [result[2] for result in results]
if np.allclose(chi_over_lam2_values, RM_TOTAL * s_grid_2000[-1], atol=0.001):
    print("PASS: chi/lam2 is constant across the sweep and equal to 10.0 within 0.1%.")
else:
    print("FAIL: chi/lam2 is not constant across the sweep or not equal to 10.0 within 0.1%.")

# Convergence check
lam2 = 0.1
K_of_s = lambda s: K_of_s_faraday(s, lam2)
S_2000 = integrate_ray(s_grid_2000, j_of_s_emission, K_of_s, S0)
S_1000 = integrate_ray(s_grid_1000, j_of_s_emission, K_of_s, S0)
chi_2000 = 0.5 * np.arctan2(S_2000[-1, 2], S_2000[-1, 1])
chi_1000 = 0.5 * np.arctan2(S_1000[-1, 2], S_1000[-1, 1])
error_ratio = abs((chi_2000 - chi_1000) / (S_2000[-1] - S_1000[-1])) * 2

if error_ratio > 16:
    print("PASS: Convergence check passed, error shrinks by ~16x when steps double.")
else:
    print("FAIL: Convergence check failed, error does not shrink by ~16x when steps double.")
```

```
import numpy as np

def integrate_ray(s_grid, j_of_s, K_of_s, S0):
    def rk4_step(S, s, h, j, K):
        k1 = h * (j(s) - K(s) @ S)
        k2 = h * (j(s + 0.5*h) - K(s + 0.5*h) @ (S + 0.5*k1))
        k3 = h * (j(s + 0.5*h) - K(s + 0.5*h) @ (S + 0.5*k2))
        k4 = h * (j(s + h) - K(s + h) @ (S + k3))
        return S + (k1 + 2*k2 + 2*k3 + k4) / 6

    S = np.zeros((len(s_grid), 4))
    S[0] = S0
    for i in range(1, len(s_grid)):
        h = s_grid[i] - s_grid[i-1]
        S[i] = rk4_step(S[i-1], s_grid[i-1], h, j_of_s, K_of_s)
    
    return S

def fit_slope(lam2_values, chi_values):
    A = np.vstack([lam2_values, np.ones(len(lam2_values))]).T
    m, c = np.linalg.lstsq(A, chi_values, rcond=None)[0]
    return m
```
